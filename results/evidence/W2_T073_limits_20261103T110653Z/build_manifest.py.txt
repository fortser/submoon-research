"""Зафиксировать evidence C131/C132 без перезаписи прежних артефактов."""
import hashlib
import importlib.metadata
import json
import platform
import sys
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parents[1]
out = root / 'results/evidence/W2_T073_limits_20261103T110653Z'
previous = json.loads((root / 'results/evidence/W2_T073_freeze_contract_20261103T103843Z/manifest.json')
                      .read_text(encoding='utf-8'))
source_names = [
    'src/submoon_research/contracts/run_manifest.py',
    'src/submoon_research/contracts/freeze.py',
    'src/submoon_research/execution.py',
    'src/submoon_research/provenance.py',
    'src/submoon_research/tracking.py',
    'tests/unit/test_provenance_batch.py',
    'tests/unit/test_provenance_execution.py',
    'tests/unit/test_run_manifest_freeze.py',
    'tests/integration/test_l1_acceptance.py',
    'scripts/analyze_w1.py',
    'scripts/acquire_geometric_states.py',
    'scripts/acquire_planetary_alignment.py',
    'scripts/verify_w0_completion.py',
    'scripts/review_w0_quality.py',
    'scripts/review_w0_independent.py',
    'reports/analysis/W2_T073_consumers_and_limits_20261103.md',
    'results/evidence/claims.csv',
]
sha = lambda data: hashlib.sha256(data).hexdigest()
sources = {}
for name in source_names:
    data = (root / name).read_bytes()
    target = out / 'sources' / (name + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as stream:
        stream.write(data)
    sources[name] = sha(data)
    if name.startswith('src/'):
        assert sources[name] == previous['sources_sha256'][name], name
assert sha((root / 'src/submoon_research/execution.py').read_bytes()) == sha((out / 'execution_before.py.txt').read_bytes())
attempts = [json.loads(line) for line in (out / 'attempts.jsonl').read_text(encoding='utf-8').splitlines()]
assert all(not row['stop_reason'] and not row['observed_survivors'] and row['returncode'] == 0 for row in attempts)
measured = sum(row['elapsed_seconds'] for row in attempts)
assert 300 + measured < 600
files = {p.relative_to(out).as_posix(): sha(p.read_bytes()) for p in sorted(out.rglob('*')) if p.is_file()}
manifest = dict(
    run_id=out.name, utc=datetime.now(timezone.utc).isoformat(), tracking_revision=718,
    status='writer_checks_passed_reader_gap_blocks_T073', checks=['W2-C131', 'W2-C132'], issue='W2-I046',
    environment=dict(python=sys.version, executable=sys.executable, platform=platform.platform(),
        packages={p: importlib.metadata.version(p) for p in ['pytest', 'pydantic', 'numpy', 'scipy', 'psutil', 'ruff']}),
    budget_prior_reserve_seconds=300, measured_new_subprocess_seconds=measured,
    accounted_subprocess_seconds=300+measured, budget_total_seconds=600,
    peak_observed_windows_rss_bytes=max(row['peak_observed_rss_bytes'] for row in attempts),
    linux_memory_limitations='Полный Linux RSS и OS memory quota не измерены/не обеспечены Windows sampling',
    sources_sha256=sources, evidence_sha256=files,
    smoke_run_id='L0-20261103T111338Z-5dce14e9', legacy_run_id='L0-20261101T184623Z-c3f2af79',
    aggregate_executed=False, independent_review=False, full_pytest=False, scientific_admission=False,
    reader_scope='src/scripts/tests; не динамические внешние consumers или historical scratch',
)
with (out / 'manifest.json').open('x', encoding='utf-8') as stream:
    json.dump(manifest, stream, ensure_ascii=False, indent=2)
print(json.dumps(dict(measured_seconds=measured, accounted_seconds=300+measured,
                     evidence_files=len(files), status=manifest['status']), ensure_ascii=False))
